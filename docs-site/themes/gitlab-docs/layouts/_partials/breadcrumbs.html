{{/* Server-renders the breadcrumb trail for the current page. Renders the trail from `functions/breadcrumbs` so crawlers and no-JS clients see real breadcrumb links in the HTML. The Vue `BreadcrumbNav` component replaces this markup on hydration. The matching BreadcrumbList structured data is emitted in the document head by `breadcrumb-schema.html`. @param Page The page object (passed as the dot context). */}} {{- $crumbs := partialCached "functions/breadcrumbs" . .RelPermalink -}} {{- $count := len $crumbs -}}
    {{- range $i, $crumb := $crumbs -}} {{- $isLast := eq (add $i 1) $count -}}
  1. {{- if $isLast -}} {{ $crumb.text }} {{- else -}} {{ $crumb.text }} / {{- end -}}
    2. {{- end -}}
