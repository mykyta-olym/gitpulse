{{/* Add the clipboard copy toolbar to code samples. Changes to the wrapper divs here should also be made in themes/gitlab-docs/layouts/_partials/archiveItem.html. A fenced code block with an `editable="true"` attribute renders as an editable code block: ```yaml {editable="true"} project: ``` Any token (name matching [A-Za-z0-9_.-]+) becomes an inline, editable field the reader can fill in before copying. See themes/gitlab-docs/src/utils/editable_code.js for the interactive behaviour. These blocks are intentionally not syntax highlighted, so the placeholder spans can be emitted without fighting Chroma's tokenizer. All other code blocks are highlighted as usual. */}} {{- if eq (.Attributes.editable | default "") "true" -}} {{- $code := trim .Inner "\n" -}} {{- $escaped := $code | htmlEscape -}} {{- $iconsSVG := "gitlab_ui/svgs-latest/icons.svg" | relURL -}} {{- $replacement := printf `<${1}>` $iconsSVG -}} {{- $withVars := replaceRE `<([A-Za-z0-9_.-]+)>` $replacement $escaped -}} 
{{ $withVars | safeHTML }}
{{- else -}}
{{- $result := transform.HighlightCodeBlock . -}} {{- $result.Wrapped -}}
{{- end -}}
